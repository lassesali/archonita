Configuration details:
- android:glEsVersion uses a 32-bit packed hexadecimal integer:
    Upper 16 bits = Major version
    Lower 16 bits = Minor version
- Reference values:
    0x00010001 -> OpenGL ES 1.1 (currently declared; line 11)
    0x00020000 -> OpenGL ES 2.0
    0x00030000 -> OpenGL ES 3.0

- Google Play & Android PackageManager use this <uses-feature> tag for device
  compatibility filtering. Note that changing AndroidManifest.xml alone only
  changes OS/Play Store filtering; it does not alter the EGL context version
  or linked NDK libraries.

IMPORTANT COMPATIBILITY WARNING:
Torque2D's Android renderer uses the fixed-function OpenGL ES 1.1 path (wrapped
via the AndroidGL2ES shim). Raising EGL_CONTEXT_CLIENT_VERSION to 2 or 3, or
switching linkage from GLESv1_CM to GLESv2/GLESv3, without rewriting the engine
rendering pipeline to programmable GLSL shaders will fail fixed-function GL
calls at compile time or runtime.

value instead of vendor/renderer/version strings. Use native logs (adprintf /
logcat) or the native/SDK runtime queries below instead.

--------------------------------------------------------------------------------
4.1 Active Context Version vs. Maximum Device-Supported Version
--------------------------------------------------------------------------------
When verifying OpenGL ES at runtime on Android, distinguish two different
questions:

1. "What OpenGL ES version is the running Torque2D EGL context using?"
   Because T2DActivity.cpp requests EGL_CONTEXT_CLIENT_VERSION = 1, calling
   glGetString(GL_VERSION) inside the active engine context reports the ES 1.1
   compatibility profile (e.g., "OpenGL ES-CM 1.1"), even on a modern GPU that
   supports OpenGL ES 3.2.

2. "What is the maximum OpenGL ES version supported by the target device?"
   To discover whether the hardware/driver supports ES 2.0, 3.0, 3.1, or 3.2:
   - Quick OS query: ActivityManager.getDeviceConfigurationInfo().reqGlEsVersion
     (fast, no EGL context needed, though some drivers/OS images only advertise
     major/minor up to 3.0 or 3.2 in system properties).
   - Authoritative GPU driver query: Create an offscreen 1x1 EGL pbuffer context
     trying ES 3 (and falling back to ES 2 / ES 1) and query GL_VERSION (and
     GL_MAJOR_VERSION / GL_MINOR_VERSION under ES 3.0+).

4.2 C++ (Android NDK - EGL + GLES)
--------------------------------------------------------------------------------
OPTION A: Querying the Active Engine Context
Place after eglMakeCurrent() in T2DActivity.cpp (or inspect in AndroidGL.cpp):

```cpp
#include <EGL/egl.h>
#include <GLES/gl.h>
#include <android/log.h>

void LogActiveTorqueGlesContext(EGLDisplay display, EGLContext context)
{
    EGLint eglClientVersion = 0;
    if (display != EGL_NO_DISPLAY && context != EGL_NO_CONTEXT) {
        eglQueryContext(display, context, EGL_CONTEXT_CLIENT_VERSION, &eglClientVersion);
    }

    const char* vendor   = reinterpret_cast<const char*>(glGetString(GL_VENDOR));
    const char* renderer = reinterpret_cast<const char*>(glGetString(GL_RENDERER));
    const char* version  = reinterpret_cast<const char*>(glGetString(GL_VERSION));

    __android_log_print(
        ANDROID_LOG_INFO, "Torque2D",
        "Active EGL client=%d | Vendor=%s | Renderer=%s | GL_VERSION=%s",
        eglClientVersion,
        vendor   ? vendor   : "(null)",
        renderer ? renderer : "(null)",
        version  ? version  : "(null)");
}
```

OPTION B: Probing Maximum Device-Supported OpenGL ES Version via Offscreen Pbuffer
Safe standalone probe using a 1x1 pbuffer surface. Preserves and restores any
currently active EGL context/surfaces, and avoids terminating the shared
EGLDisplay if an engine context is already active.

```cpp
#include <EGL/egl.h>
#include <EGL/eglext.h>
#include <GLES/gl.h>
#include <android/log.h>

struct GlesProbeResult {
    int eglClientMajor;       // Highest EGL context client version created (3, 2, or 1)
    int glMajor;              // GL_MAJOR_VERSION (when >= 3) or parsed fallback
    int glMinor;              // GL_MINOR_VERSION (when >= 3) or parsed fallback
    const char* versionStr;   // Driver GL_VERSION string (static driver storage while initialized)
    const char* rendererStr;  // Driver GL_RENDERER string
};

GlesProbeResult ProbeMaxSupportedGlesVersion()
{
    GlesProbeResult result = {0, 0, 0, nullptr, nullptr};

    EGLDisplay prevDisplay = eglGetCurrentDisplay();
    EGLSurface prevDraw    = eglGetCurrentSurface(EGL_DRAW);
    EGLSurface prevRead    = eglGetCurrentSurface(EGL_READ);
    EGLContext prevContext = eglGetCurrentContext();

    EGLDisplay display = (prevDisplay != EGL_NO_DISPLAY)
        ? prevDisplay
        : eglGetDisplay(EGL_DEFAULT_DISPLAY);
    if (display == EGL_NO_DISPLAY) {
        return result;
    }

    const bool hadActiveContext = (prevContext != EGL_NO_CONTEXT);
    if (!hadActiveContext) {
        if (!eglInitialize(display, nullptr, nullptr)) {
            return result;
        }
    }

    // Try ES 3 first (via ES2-compatible config or ES3 bit), then fall back to 2 and 1.
    const int candidates[] = {3, 2, 1};
    for (int ver : candidates) {
        const EGLint renderableBit = (ver >= 2) ? EGL_OPENGL_ES2_BIT : EGL_OPENGL_ES_BIT;
        const EGLint cfgAttribs[] = {
            EGL_SURFACE_TYPE,    EGL_PBUFFER_BIT,
            EGL_RENDERABLE_TYPE, renderableBit,
            EGL_RED_SIZE,        8,
            EGL_GREEN_SIZE,      8,
            EGL_BLUE_SIZE,       8,
            EGL_NONE
        };

        EGLConfig config = nullptr;
        EGLint numConfigs = 0;
        if (!eglChooseConfig(display, cfgAttribs, &config, 1, &numConfigs) || numConfigs == 0) {
            continue;
        }

        const EGLint pbufferAttribs[] = {
            EGL_WIDTH,  1,
            EGL_HEIGHT, 1,
            EGL_NONE
        };
        EGLSurface pbuffer = eglCreatePbufferSurface(display, config, pbufferAttribs);
        if (pbuffer == EGL_NO_SURFACE) {
            continue;
        }

        const EGLint ctxAttribs[] = {
            EGL_CONTEXT_CLIENT_VERSION, ver,
            EGL_NONE
        };
        EGLContext probeCtx = eglCreateContext(display, config, EGL_NO_CONTEXT, ctxAttribs);
        if (probeCtx == EGL_NO_CONTEXT) {
            eglDestroySurface(display, pbuffer);
            continue;
        }

        if (eglMakeCurrent(display, pbuffer, pbuffer, probeCtx)) {
            result.eglClientMajor = ver;
            result.versionStr     = reinterpret_cast<const char*>(glGetString(GL_VERSION));
            result.rendererStr    = reinterpret_cast<const char*>(glGetString(GL_RENDERER));

            if (ver >= 3) {
                // GL_MAJOR_VERSION (0x821B) and GL_MINOR_VERSION (0x821C) exist in ES 3.0+
                GLint maj = 0, min = 0;
                glGetIntegerv(0x821B, &maj);
                glGetIntegerv(0x821C, &min);
                if (glGetError() == GL_NO_ERROR && maj > 0) {
                    result.glMajor = maj;
                    result.glMinor = min;
                }
            }
            if (result.glMajor == 0) {
                result.glMajor = ver;
                result.glMinor = (ver == 1) ? 1 : 0;
            }

            __android_log_print(
                ANDROID_LOG_INFO, "Torque2D",
                "Max Supported GLES Probe: client=%d (%d.%d) | Renderer=%s | Version=%s",
                result.eglClientMajor, result.glMajor, result.glMinor,
                result.rendererStr ? result.rendererStr : "(null)",
                result.versionStr  ? result.versionStr  : "(null)");

            // Restore prior context cleanly before tearing down probe resources.
            if (hadActiveContext) {
                eglMakeCurrent(prevDisplay, prevDraw, prevRead, prevContext);
            } else {
                eglMakeCurrent(display, EGL_NO_SURFACE, EGL_NO_SURFACE, EGL_NO_CONTEXT);
            }
            eglDestroyContext(display, probeCtx);
            eglDestroySurface(display, pbuffer);
            break;
        }

        eglDestroyContext(display, probeCtx);
        eglDestroySurface(display, pbuffer);
    }

    // Never terminate the shared EGLDisplay while an engine context is live.
    if (!hadActiveContext) {
        eglTerminate(display);
    }
    return result;
}
```

4.3 Java (Android SDK - ActivityManager & EGL14 Offscreen Probe)
--------------------------------------------------------------------------------
Can be placed inside MyNativeActivity.java or T2DUtilities.java under:
  engine/compilers/android-studio/app/src/main/java/com/garagegames/torque2d/

Note: Call the EGL14 probe during Activity startup before native engine EGL
initialization (or skip eglTerminate() if the native thread has already
initialized EGL_DEFAULT_DISPLAY).

```java
package com.garagegames.torque2d;

import android.app.ActivityManager;
import android.content.Context;
import android.content.pm.ConfigurationInfo;
import android.opengl.EGL14;
import android.opengl.EGLConfig;
import android.opengl.EGLContext;
import android.opengl.EGLDisplay;
import android.opengl.EGLSurface;
import android.opengl.GLES20;
import android.opengl.GLES30;
import android.util.Log;

public final class GlEsVersionHelper {
    private static final String TAG = "Torque2D";

    private GlEsVersionHelper() {}

    /**
     * Fast OS-reported check via ActivityManager ConfigurationInfo.
     * Returns packed hex version (e.g. 0x00030002 -> "3.2").
     */
    public static int getOsReportedGlEsVersion(Context context) {
        ActivityManager am = (ActivityManager) context.getSystemService(Context.ACTIVITY_SERVICE);
        if (am == null) {
            return ConfigurationInfo.GL_ES_VERSION_UNDEFINED;
        }
        ConfigurationInfo info = am.getDeviceConfigurationInfo();
        int packed = info.reqGlEsVersion;
        int major = (packed & 0xffff0000) >> 16;
        int minor = packed & 0x0000ffff;
        Log.i(TAG, "ConfigurationInfo reqGlEsVersion=" + info.getGlEsVersion()
                + " (" + major + "." + minor + ", hex=0x" + Integer.toHexString(packed) + ")");
        return packed;
    }

    /**
     * Exact GPU driver query using an offscreen 1x1 EGL14 pbuffer context.
     * Proves whether ES 3.x (including 3.0 / 3.1 / 3.2) or ES 2.0 works on the GPU.
     */
    public static String queryExactGpuGlEsVersionString() {
        EGLDisplay prevDisplay = EGL14.eglGetCurrentDisplay();
        EGLSurface prevDraw    = EGL14.eglGetCurrentSurface(EGL14.EGL_DRAW);
        EGLSurface prevRead    = EGL14.eglGetCurrentSurface(EGL14.EGL_READ);
        EGLContext prevContext = EGL14.eglGetCurrentContext();

        boolean hadActiveContext = prevContext != null && !prevContext.equals(EGL14.EGL_NO_CONTEXT);
        EGLDisplay display = (prevDisplay != null && !prevDisplay.equals(EGL14.EGL_NO_DISPLAY))
                ? prevDisplay
                : EGL14.eglGetDisplay(EGL14.EGL_DEFAULT_DISPLAY);
        if (display == null || display.equals(EGL14.EGL_NO_DISPLAY)) {
            return null;
        }

        if (!hadActiveContext) {
            int[] eglVer = new int[2];
            if (!EGL14.eglInitialize(display, eglVer, 0, eglVer, 1)) {
                return null;
            }
        }

        String detectedVersion = null;
        int[] candidates = new int[]{3, 2};
        for (int clientVer : candidates) {
            int[] cfgAttribs = new int[]{
                    EGL14.EGL_SURFACE_TYPE, EGL14.EGL_PBUFFER_BIT,
                    EGL14.EGL_RENDERABLE_TYPE, EGL14.EGL_OPENGL_ES2_BIT,
                    EGL14.EGL_RED_SIZE, 8,
                    EGL14.EGL_GREEN_SIZE, 8,
                    EGL14.EGL_BLUE_SIZE, 8,
                    EGL14.EGL_NONE
            };
            EGLConfig[] configs = new EGLConfig[1];
            int[] numConfigs = new int[1];
            if (!EGL14.eglChooseConfig(display, cfgAttribs, 0, configs, 0, 1, numConfigs, 0)
                    || numConfigs[0] == 0) {
                continue;
            }

            int[] pbufAttribs = new int[]{
                    EGL14.EGL_WIDTH, 1,
                    EGL14.EGL_HEIGHT, 1,
                    EGL14.EGL_NONE
            };
            EGLSurface pbuffer = EGL14.eglCreatePbufferSurface(display, configs[0], pbufAttribs, 0);
            if (pbuffer == null || pbuffer.equals(EGL14.EGL_NO_SURFACE)) {
                continue;
            }

            int[] ctxAttribs = new int[]{
                    EGL14.EGL_CONTEXT_CLIENT_VERSION, clientVer,
                    EGL14.EGL_NONE
            };
            EGLContext ctx = EGL14.eglCreateContext(
                    display, configs[0], EGL14.EGL_NO_CONTEXT, ctxAttribs, 0);
            if (ctx == null || ctx.equals(EGL14.EGL_NO_CONTEXT)) {
                EGL14.eglDestroySurface(display, pbuffer);
                continue;
            }

            if (EGL14.eglMakeCurrent(display, pbuffer, pbuffer, ctx)) {
                String versionStr  = GLES20.glGetString(GLES20.GL_VERSION);
                String vendorStr   = GLES20.glGetString(GLES20.GL_VENDOR);
                String rendererStr = GLES20.glGetString(GLES20.GL_RENDERER);

                if (clientVer >= 3) {
                    int[] major = new int[1];
                    int[] minor = new int[1];
                    GLES30.glGetIntegerv(GLES30.GL_MAJOR_VERSION, major, 0);
                    GLES30.glGetIntegerv(GLES30.GL_MINOR_VERSION, minor, 0);
                    Log.i(TAG, "Exact GPU GLES=" + major[0] + "." + minor[0]
                            + " | Vendor=" + vendorStr
                            + " | Renderer=" + rendererStr
                            + " | GL_VERSION=" + versionStr);
                } else {
                    Log.i(TAG, "Exact GPU GLES=2.0 | Vendor=" + vendorStr
                            + " | Renderer=" + rendererStr
                            + " | GL_VERSION=" + versionStr);
                }
                detectedVersion = versionStr;

                if (hadActiveContext) {
                    EGL14.eglMakeCurrent(prevDisplay, prevDraw, prevRead, prevContext);
                } else {
                    EGL14.eglMakeCurrent(
                            display, EGL14.EGL_NO_SURFACE, EGL14.EGL_NO_SURFACE, EGL14.EGL_NO_CONTEXT);
                }
                EGL14.eglDestroyContext(display, ctx);
                EGL14.eglDestroySurface(display, pbuffer);
                break;
            }

            EGL14.eglDestroyContext(display, ctx);
            EGL14.eglDestroySurface(display, pbuffer);
        }

        if (!hadActiveContext) {
            EGL14.eglTerminate(display);
        }
        return detectedVersion;
    }
}
```

```kotlin
package com.garagegames.torque2d

import android.app.ActivityManager
import android.content.Context
import android.opengl.EGL14
import android.opengl.EGLConfig
import android.opengl.GLES20
import android.opengl.GLES30
import android.util.Log

data class GlesDeviceInfo(
    val osReportedPacked: Int,
    val exactMajor: Int,
    val exactMinor: Int,
    val versionString: String?,
    val rendererString: String?,
    val vendorString: String?
)

object GlEsVersionHelperKt {
    private const val TAG = "Torque2D"

    fun query(context: Context): GlesDeviceInfo {
        val am = context.getSystemService(Context.ACTIVITY_SERVICE) as? ActivityManager
        val osPacked = am?.deviceConfigurationInfo?.reqGlEsVersion ?: 0

        val prevDisplay = EGL14.eglGetCurrentDisplay()
        val prevDraw = EGL14.eglGetCurrentSurface(EGL14.EGL_DRAW)
        val prevRead = EGL14.eglGetCurrentSurface(EGL14.EGL_READ)
        val prevContext = EGL14.eglGetCurrentContext()
        val hadActiveContext = prevContext != null && prevContext != EGL14.EGL_NO_CONTEXT

        val display = if (prevDisplay != null && prevDisplay != EGL14.EGL_NO_DISPLAY) {
            prevDisplay
        } else {
            EGL14.eglGetDisplay(EGL14.EGL_DEFAULT_DISPLAY)
        }
        if (display == null || display == EGL14.EGL_NO_DISPLAY) {
            return GlesDeviceInfo(osPacked, (osPacked ushr 16), osPacked and 0xFFFF, null, null, null)
        }

        if (!hadActiveContext) {
            val ver = IntArray(2)
            if (!EGL14.eglInitialize(display, ver, 0, ver, 1)) {
                return GlesDeviceInfo(osPacked, (osPacked ushr 16), osPacked and 0xFFFF, null, null, null)
            }
        }

        var exactMajor = 0
        var exactMinor = 0
        var versionStr: String? = null
        var rendererStr: String? = null
        var vendorStr: String? = null

        for (clientVer in intArrayOf(3, 2)) {
            val cfgAttribs = intArrayOf(
                EGL14.EGL_SURFACE_TYPE, EGL14.EGL_PBUFFER_BIT,
                EGL14.EGL_RENDERABLE_TYPE, EGL14.EGL_OPENGL_ES2_BIT,
                EGL14.EGL_RED_SIZE, 8,
                EGL14.EGL_GREEN_SIZE, 8,
                EGL14.EGL_BLUE_SIZE, 8,
                EGL14.EGL_NONE
            )
            val configs = arrayOfNulls<EGLConfig>(1)
            val numConfigs = IntArray(1)
            if (!EGL14.eglChooseConfig(display, cfgAttribs, 0, configs, 0, 1, numConfigs, 0)
                || numConfigs[0] == 0
            ) {
                continue
            }

            val config = configs[0] ?: continue
            val pbuffer = EGL14.eglCreatePbufferSurface(
                display, config, intArrayOf(EGL14.EGL_WIDTH, 1, EGL14.EGL_HEIGHT, 1, EGL14.EGL_NONE), 0
            )
            if (pbuffer == null || pbuffer == EGL14.EGL_NO_SURFACE) continue

            val ctx = EGL14.eglCreateContext(
                display, config, EGL14.EGL_NO_CONTEXT,
                intArrayOf(EGL14.EGL_CONTEXT_CLIENT_VERSION, clientVer, EGL14.EGL_NONE), 0
            )
            if (ctx == null || ctx == EGL14.EGL_NO_CONTEXT) {
                EGL14.eglDestroySurface(display, pbuffer)
                continue
            }

            if (EGL14.eglMakeCurrent(display, pbuffer, pbuffer, ctx)) {
                versionStr = GLES20.glGetString(GLES20.GL_VERSION)
                rendererStr = GLES20.glGetString(GLES20.GL_RENDERER)
                vendorStr = GLES20.glGetString(GLES20.GL_VENDOR)

                if (clientVer >= 3) {
                    val maj = IntArray(1)
                    val min = IntArray(1)
                    GLES30.glGetIntegerv(GLES30.GL_MAJOR_VERSION, maj, 0)
                    GLES30.glGetIntegerv(GLES30.GL_MINOR_VERSION, min, 0)
                    exactMajor = maj[0]
                    exactMinor = min[0]
                } else {
                    exactMajor = 2
                    exactMinor = 0
                }

                Log.i(TAG, "GLES probe=$exactMajor.$exactMinor | OS=0x${osPacked.toString(16)} | $versionStr")

                if (hadActiveContext) {
                    EGL14.eglMakeCurrent(prevDisplay, prevDraw, prevRead, prevContext)
                } else {
                    EGL14.eglMakeCurrent(display, EGL14.EGL_NO_SURFACE, EGL14.EGL_NO_SURFACE, EGL14.EGL_NO_CONTEXT)
                }
                EGL14.eglDestroyContext(display, ctx)
                EGL14.eglDestroySurface(display, pbuffer)
                break
            }

            EGL14.eglDestroyContext(display, ctx)
            EGL14.eglDestroySurface(display, pbuffer)
        }

        if (!hadActiveContext) {
            EGL14.eglTerminate(display)
        }
        return GlesDeviceInfo(osPacked, exactMajor, exactMinor, versionStr, rendererStr, vendorStr)
    }
}
```
